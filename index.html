PETICIÓN POST DESDE LA LANDING HACIA TU API EN .NET async function handleFormSubmit(event) { event.preventDefault(); const name = document.getElementById('lead-name').value; const phone = document.getElementById('lead-phone').value; const email = document.getElementById('lead-email').value; const complex = document.getElementById('lead-complex').value; const plan = document.getElementById('lead-plan').value; // URL PÚBLICA DE TU API (localtunnel) const API_URL = "https://olive-mice-chew.loca.lt/api/Suscripciones"; try { const response = await fetch(`${API_URL}/inscribir`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ nombre: name, telefono: phone, correo: email, nombreComplejo: complex, plan: plan }) }); if (response.ok) { document.getElementById('res-user-name').innerText = name; document.getElementById('lead-form').classList.add('hidden'); document.getElementById('success-message').classList.remove('hidden'); } else { const error = await response.json(); alert('Error: ' + (error.mensaje || 'No se pudo procesar la inscripción')); } } catch (err) { console.error("Error:", err); alert('Error de conexión con el backend. Asegúrate de que el servidor esté corriendo.'); } } // 2. PETICIÓN PUT DESDE LA LANDING PARA ATENDER INSCRIPCIÓN async function atenderInscripcionDesdeLanding() { const email = document.getElementById('lead-email').value; const statusText = document.getElementById('atender-status'); // URL PÚBLICA DE TU API (localtunnel) const API_URL = "https://olive-mice-chew.loca.lt/api/Suscripciones"; statusText.innerText = "Enviando solicitud PUT..."; try { const res = await fetch(`${API_URL}/atender?contacto=${encodeURIComponent(email)}`, { method: 'PUT' }); if (res.ok) { statusText.className = "text-xs text-brand-neon font-bold mt-2"; statusText.innerText = "¡Éxito! Estado actualizado a 'Atendida' en la base de datos."; } else { statusText.className = "text-xs text-red-400 font-bold mt-2"; statusText.innerText = "No se encontró el registro en Supabase."; } } catch (e) { statusText.className = "text-xs text-red-400 mt-2"; statusText.innerText = "Error: Verifica que el servidor esté corriendo."; } }
